2 Листинг xUnit и FluentAssertions

      Листинг           —    Unit-тест     на        два       диапазонных              значения,
ReadingServiceTests.cs
 [Theory]
 [InlineData(-0.01)]
 [InlineData(1000000.01)]
 public void OutOfRangeReading_ShouldBeRejected(double value)
 {
     var result = NewService().Create(new("METER-001", (decimal)value,
          DateTimeOffset.UtcNow));
     result.Status.Should().Be(CreateStatus.Invalid);
 }
      Листинг — Интеграционный тест запрета Swagger в Production
 [Fact]
 public async Task ProductionSwagger_ShouldNotBeAvailable()
 {
     using var factory = new TestApp(); // Environment = Production
     using var client = factory.CreateClient(new()
     {
         BaseAddress = new Uri("https://localhost"),
         AllowAutoRedirect = false
     });
     var response = await client.GetAsync("/swagger/v1/swagger.json");
     response.StatusCode.Should().Be(HttpStatusCode.NotFound);
 }
